Conexión con Google Sheets

Prueba una URL temporalmente o copia el código Apps Script

💡 Para que les funcione a todos en GitHub: Pega esta misma URL dentro de GOOGLE_SCRIPT_URL: "" al inicio del archivo index.html antes de subirlo.

Código Google Apps Script (Compatible con esta App) 
const SHEET_ASESORES = "Asesores";
const SHEET_REPORTES = "Reportes";

function doGet(e) { return handleRequest(e); }
function doPost(e) { return handleRequest(e); }

function handleRequest(e) {
  const lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    inicializarHojas(ss);
    const action = (e && e.parameter && e.parameter.action) || "sync";
    let payload = {};
    if (e && e.postData && e.postData.contents) {
      try { payload = JSON.parse(e.postData.contents); } catch (err) {}
    }
    const finalAction = payload.action || action;

    if (finalAction === "register_advisor" || finalAction === "update_advisor") {
      const sheet = ss.getSheetByName(SHEET_ASESORES);
      const data = sheet.getDataRange().getValues();
      const phone = String(payload.phone || "").replace(/\D/g, "");
      const name = String(payload.name || "").trim();
      const metaMensual = Number(payload.metaMensual) || 15000000;
      const metaConvencion = Number(payload.metaConvencion) || 120000000;
      if (!phone || !name) return jsonResponse({ status: "error", message: "Faltan datos" });

      let foundRow = -1;
      for (let i = 1; i < data.length; i++) {
        if (String(data[i][0]).replace(/\D/g, "") === phone) { foundRow = i + 1; break; }
      }
      if (foundRow > -1) {
        sheet.getRange(foundRow, 2).setValue(name);
        sheet.getRange(foundRow, 3).setValue(metaMensual);
        sheet.getRange(foundRow, 4).setValue(metaConvencion);
      } else {
        sheet.appendRow([phone, name, metaMensual, metaConvencion, new Date()]);
      }
      return jsonResponse({ status: "ok", data: obtenerTodo(ss) });
    }

    if (finalAction === "add_report") {
      const sheet = ss.getSheetByName(SHEET_REPORTES);
      const id = payload.id || ("REP-" + new Date().getTime());
      const fecha = payload.fecha || Utilities.formatDate(new Date(), "America/Bogota", "yyyy-MM-dd");
      const phone = String(payload.phone || "").replace(/\D/g, "");
      const asesor = String(payload.asesor || "");
      const cliente = String(payload.cliente || "Cliente General");
      const ramo = String(payload.ramo || "Otros");
      const valor = Number(payload.valor) || 0;
      const polizas = Number(payload.polizas) || 1;
      sheet.appendRow([id, fecha, phone, asesor, cliente, ramo, valor, polizas, new Date()]);
      return jsonResponse({ status: "ok", data: obtenerTodo(ss) });
    }

    if (finalAction === "delete_report") {
      const sheet = ss.getSheetByName(SHEET_REPORTES);
      const data = sheet.getDataRange().getValues();
      const targetId = String(payload.id || "");
      for (let i = 1; i < data.length; i++) {
        if (String(data[i][0]) === targetId) { sheet.deleteRow(i + 1); break; }
      }
      return jsonResponse({ status: "ok", data: obtenerTodo(ss) });
    }

    return jsonResponse({ status: "ok", data: obtenerTodo(ss) });
  } catch (error) {
    return jsonResponse({ status: "error", message: error.toString() });
  } finally {
    lock.releaseLock();
  }
}

function inicializarHojas(ss) {
  let sA = ss.getSheetByName(SHEET_ASESORES);
  if (!sA) {
    sA = ss.insertSheet(SHEET_ASESORES);
    sA.appendRow(["Telefono", "Nombre", "MetaMensual", "MetaConvencion", "FechaRegistro"]);
    sA.getRange("A1:E1").setFontWeight("bold").setBackground("#0f172a").setFontColor("#ffffff");
  }
  let sR = ss.getSheetByName(SHEET_REPORTES);
  if (!sR) {
    sR = ss.insertSheet(SHEET_REPORTES);
    sR.appendRow(["ID", "Fecha", "Telefono", "Asesor", "Cliente", "Ramo", "Valor", "Polizas", "Timestamp"]);
    sR.getRange("A1:I1").setFontWeight("bold").setBackground("#0f172a").setFontColor("#ffffff");
  }
}

function obtenerTodo(ss) {
  const rawA = ss.getSheetByName(SHEET_ASESORES).getDataRange().getValues();
  const asesores = [];
  for (let i = 1; i < rawA.length; i++) {
    if (rawA[i][0]) {
      asesores.push({
        phone: String(rawA[i][0]).replace(/\D/g, ""),
        name: String(rawA[i][1]),
        metaMensual: Number(rawA[i][2]) || 15000000,
        metaConvencion: Number(rawA[i][3]) || 120000000
      });
    }
  }
  const rawR = ss.getSheetByName(SHEET_REPORTES).getDataRange().getValues();
  const reportes = [];
  for (let i = 1; i < rawR.length; i++) {
    if (rawR[i][0]) {
      let f = rawR[i][1];
      if (f instanceof Date) f = Utilities.formatDate(f, "America/Bogota", "yyyy-MM-dd");
      reportes.push({
        id: String(rawR[i][0]), fecha: String(f),
        phone: String(rawR[i][2]).replace(/\D/g, ""), asesor: String(rawR[i][3]),
        cliente: String(rawR[i][4]), ramo: String(rawR[i][5]),
        valor: Number(rawR[i][6]) || 0, polizas: Number(rawR[i][7]) || 1
      });
    }
  }
  return { asesores: asesores, reportes: reportes };
}

function jsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
